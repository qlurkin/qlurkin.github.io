Le Problème

{

    if (string.IsNullOrWhiteSpace(text))
    {
        StatusText.Text = "Veuillez saisir une note.";
        return;
    }

    _noteManager.Add(text);
    NoteTextBox.Text = "";

    StatusText.Text = "Note ajoutée.";

  • La classe MainWindow mélange toujours plusieurs responsabilités:
    • elle récupère le texte depuis un contrôle graphique.
    • elle vérifie si la saisie est valide.
    • elle demande au gestionnaire d’ajouter une note.
    • elle vide le champ de saisie.
    • elle modifie le message affiché à l’utilisateur.
  • Certaines opérations concernent les données et les règles de gestion; d’autres concernent l’interface.

Le ViewModel

  • le ViewModel permet de séparer ce que l’interface doit afficher et faire de la manière dont cette interface est construite.

  • Le ViewModel s’occupe de l’état propre à l’écran

    • Quel texte est actuellement saisi ?
    • Quelle tâche est sélectionnée ?
    • Quel message faut-il afficher ?
    • Que doit-il se passer lorsque l’utilisateur demande l’ajout d’une tâche ?

  • Ces informations ne sont pas vraiment des règles métier. Elles concernent la manière dont l’utilisateur interagit avec l’application.

  • Le ViewModel expose à l’interface les données et les actions dont elle a besoin.

Le ViewModel - Concrètement

  • Il est utilisé comment DataContext de la vue

  • Il offre des propriété que la vue peut observer (Binding) et modifier (Two Way Binding)

  • Le ViewModel ne connait pas la vue. Il ne peut pas accéder aux contrôles (widgets)

    Le code suivant de notre gestionnaire de click:

    string text = NoteTextBox.Text ?? "";

    deviendra une propriété dans le ViewModel:

    public string NewNoteText { get; set; } = "";

    Et la vue l’utilisera:

    <TextBox Text="{Binding NewNoteText, Mode=TwoWay}"
       PlaceholderText="Saisissez une note..." />

Le ViewModel - Action

  • Le ViewModel peut ensuite proposer une action :
public void AddNote()

    if (string.IsNullOrWhiteSpace(NewNoteText))
    {
        StatusMessage = "Veuillez saisir une note.";
        return;
    }

    manager.Add(NewNoteText);

Le ViewModel - Tests

  • Sans ViewModel, pour tester l’ajout d’une tâche, on pourrait être obligé de lancer la fenêtre et de simuler des clics.

  • Avec un ViewModel, on peut tester directement :

var viewModel = new MainWindowViewModel();

viewModel.NewNoteText = "Préparer le cours";
viewModel.AddNote();

Console.WriteLine(viewModel.manager.NoteCount);
Console.WriteLine(viewModel.StatusMessage);
  • la logique de présentation peut être testée indépendamment de l’interface graphique.

using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace Notes;

public class MainWindowViewModel : INotifyPropertyChanged
{
    public NoteManager Manager { get; } = new();

    private string newNoteText = "";

    public string NewNoteText
    {
        get => newNoteText;
        set
        {
            if (newNoteText == value)
                return;

            newNoteText = value;
            OnPropertyChanged();
        }
    }

    private Note? selectedNote;

    public Note? SelectedNote
    {
        get => selectedNote;
        set
        {
            if (selectedNote == value)
                return;

            selectedNote = value;
            OnPropertyChanged();
        }
    }

    private string statusMessage = "Prêt.";

    public string StatusMessage
    {
        get => statusMessage;
        set
        {
            if (statusMessage == value)
                return;

            statusMessage = value;
            OnPropertyChanged();
        }
    }

    public void AddNote()
    {
        if (string.IsNullOrWhiteSpace(NewNoteText))
        {
            StatusMessage = "Veuillez saisir une note.";
            return;
        }

        Manager.Add(NewNoteText);

        NewNoteText = "";
        StatusMessage = "Note ajoutée.";
    }

    public void DeleteSelectedNote()
    {
        if (SelectedNote == null)
        {
            StatusMessage = "Veuillez sélectionner une note.";
            return;
        }

        Manager.Remove(SelectedNote);
        SelectedNote = null;
        StatusMessage = "Tâche supprimée.";
    }

    public event PropertyChangedEventHandler? PropertyChanged;

    // Permet de recevoir le nom de la propriété appelante automatiquement
    protected void OnPropertyChanged([CallerMemberName] string? propertyName = null)
    {
        PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(propertyName));
    }
}

Les Commandes

  • Dommage qu’on ne puisse pas directement relier les méthodes du ViewModel aux contrôles du XAML…

  • Utilisons les ICommand de .NET !

  • ICommand est une interface qui correspond à ceci:

public interface ICommand
{
    event EventHandler? CanExecuteChanged;

    bool CanExecute(object? parameter);

    void Execute(object? parameter);
}
  • On peut les lier dans le XAML:
<Button Content="Ajouter"
        Command="{Binding AddNoteCommand}" />

RelayCommand

using System;
using System.Windows.Input;

namespace Notes;

// ICommand est une interface, il nous faut une implémentation
public class RelayCommand : ICommand
{
    private readonly Action execute;
    private readonly Func<bool>? canExecute;

    public RelayCommand(
        Action execute,
        Func<bool>? canExecute = null)
    {
        this.execute = execute;
        this.canExecute = canExecute;
    }

    public bool CanExecute(object? parameter)
    {
        return canExecute == null || canExecute();
    }

    public void Execute(object? parameter)
    {
        execute();
    }

    public event EventHandler? CanExecuteChanged;

    public void RaiseCanExecuteChanged()
    {
        CanExecuteChanged?.Invoke(this, EventArgs.Empty);
    }
}

Commandes dans le ViewModel

using System.ComponentModel;
using System.Runtime.CompilerServices;

namespace Notes;

public class MainWindowViewModel : INotifyPropertyChanged
{
    public NoteManager Manager { get; } = new();

    // Propriétés des commandes
    public RelayCommand AddNoteCommand { get; }
    public RelayCommand DeleteSelectedNoteCommand { get; }

    private string newNoteText = "";

    public string NewNoteText
    {
        get => newNoteText;
        set
        {
            if (newNoteText == value)
                return;

            newNoteText = value;
            OnPropertyChanged();
        }
    }

    private Note? selectedNote;

    public Note? SelectedNote
    {
        get => selectedNote;
        set
        {
            if (selectedNote == value)
                return;

            selectedNote = value;
            OnPropertyChanged();
            // Le CanExecute de la commande DeleteSelectedNoteCommand
            // doit être réévalué
            DeleteSelectedNoteCommand.RaiseCanExecuteChanged();
        }
    }

    private string statusMessage = "Prêt.";

    public string StatusMessage
    {
        get => statusMessage;
        set
        {
            if (statusMessage == value)
                return;

            statusMessage = value;
            OnPropertyChanged();
        }
    }

    public MainWindowViewModel()
    {
        // Initialisation des commandes
        AddNoteCommand = new RelayCommand(AddNote);

        DeleteSelectedNoteCommand = new RelayCommand(
            DeleteSelectedNote,
            () => SelectedNote != null);
    }

    public void AddNote()
    {
        if (string.IsNullOrWhiteSpace(NewNoteText))
        {
            StatusMessage = "Veuillez saisir une note.";
            return;
        }

        Manager.Add(NewNoteText);

        NewNoteText = "";
        StatusMessage = "Note ajoutée.";
    }

    public void DeleteSelectedNote()
    {
        if (SelectedNote == null)
        {
            StatusMessage = "Veuillez sélectionner une note.";
            return;
        }

        Manager.Remove(SelectedNote);
        SelectedNote = null;
        StatusMessage = "Tâche supprimée.";
    }

    public event PropertyChangedEventHandler? PropertyChanged;

    protected void OnPropertyChanged(
        [CallerMemberName] string? propertyName = null)
    {
        PropertyChanged?.Invoke(
            this,
            new PropertyChangedEventArgs(propertyName));
    }
}

Distinction importante

  • AddNote() est une méthode qui exécute une opération.
  • AddNoteCommand est un objet qui représente cette action pour l’interface, avec la possibilité de déterminer si elle est autorisée.

Le pattern MVVM

  • Model View ViewModel
  • Pattern d’architecture software
  • 3 parties:
    • Model: stocke et gère les données, ne connait rien de la View ou du ViewModel
    • View: représente l’interface utilisateur, interagit avec le ViewModel
    • ViewModel: est entre le Model et la View, ne connait rien de la View, interagit avec le Model

MainWindowViewModel avec CommunityToolkit.Mvvm

using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;

namespace Notes;

// La classe doit être partial
public partial class MainWindowViewModel : ObservableObject
{
    public NoteManager Manager { get; } = new();

    // La propriété publique observable sera générée automatiquement
    [ObservableProperty]
    private string newNoteText = "";

    // Cette propriété doit notifier que le CanExecute
    // de DeleteSelectedNoteCommand doit être réévalué
    [ObservableProperty]
    [NotifyCanExecuteChangedFor(nameof(DeleteSelectedNoteCommand))]
    private Note? selectedNote;

    [ObservableProperty]
    private string statusMessage = "Prêt.";

    // On peut supprimer notre implémentation de RelayCommand
    // car il existe déjà dans CommunityToolkit.Mvvm
    // La création de la commande sera faite automatiquement
    [RelayCommand]
    public void AddNote()
    {
        if (string.IsNullOrWhiteSpace(NewNoteText))
        {
            StatusMessage = "Veuillez saisir une note.";
            return;
        }

        Manager.Add(NewNoteText);

        NewNoteText = "";
        StatusMessage = "Note ajoutée.";
    }

    private bool CanDeleteNote() => SelectedNote != null;

    // Ici avec un CanExecute
    [RelayCommand (CanExecute = nameof(CanDeleteNote))]
    public void DeleteSelectedNote()
    {
        if (SelectedNote == null)
        {
            StatusMessage = "Veuillez sélectionner une note.";
            return;
        }

        Manager.Remove(SelectedNote);
        SelectedNote = null;
        StatusMessage = "Tâche supprimée.";
    }
}

NoteManager avec CommunityToolkit.Mvvm

using System.Collections.ObjectModel;
using CommunityToolkit.Mvvm.ComponentModel;

namespace demo;

public partial class NoteManager  : ObservableObject
{
    private ObservableCollection<Note> notes { get; } = new();

    public ReadOnlyObservableCollection<Note> Notes { get; }

    public int NoteCount
    {
        get
        {
            return Notes.Count;
        }
    }

    public NoteManager()
    {
        Notes = new ReadOnlyObservableCollection<Note>(notes);

        // Subscription to the change of the collection
        notes.CollectionChanged += (_, _) =>
        {
            OnPropertyChanged(nameof(NoteCount));
        };
    }

    public void Add(string text)
    {
        notes.Add(new Note(text));
    }

    public void Remove(Note note) {
        notes.Remove(note);
    }
}
